Necesito un panel para configurar los valores de la IP, API_ENDPOINT, API_KEY, habilitar antenas, modificar las ganancias de cada una de las antenas, decimal o hexadecimal, el RSSI de corte etc

Voy a crear un panel de configuración con interfaz gráfica usando JavaFX para controlar todos los parámetros del lector RFID.

/**
 * Panel de configuración para el lector RFID Zebra FX7500
 * Permite configurar todos los parámetros antes de iniciar la lectura
 */

    private TextField portTextField;

    private PasswordField apiKeyField;
    private CheckBox apiKeyVisibleCheckbox;

    // Componentes de formato EPC

    private RadioButton decimalRadio;

    @Override
    public void start(Stage primaryStage) {
        primaryStage.setTitle("Panel de Configuración - Zebra FX7500 RFID");

        // Layout principal
        BorderPane mainLayout = new BorderPane();
        mainLayout.setPadding(new Insets(10));

        // Crear tabs para organizar configuraciones
        TabPane tabPane = new TabPane();
        tabPane.setTabClosingPolicy(TabPane.TabClosingPolicy.UNAVAILABLE);

        // Tab 1: Configuración de Conexión
        Tab connectionTab = new Tab("Conexión");
        connectionTab.setContent(createConnectionPanel());

        // Tab 2: Configuración de API
        Tab apiTab = new Tab("API");
        apiTab.setContent(createAPIPanel());

        // Tab 3: Configuración de Antenas
        Tab antennasTab = new Tab("Antenas");
        antennasTab.setContent(createAntennasPanel());

        // Tab 4: Configuración Avanzada
        Tab advancedTab = new Tab("Avanzado");
        advancedTab.setContent(createAdvancedPanel());

        tabPane.getTabs().addAll(connectionTab, apiTab, antennasTab, advancedTab);

        // Panel de botones
        HBox buttonPanel = createButtonPanel();

        // Área de status
        statusArea = new TextArea();
        statusArea.setEditable(false);
        statusArea.setPrefRowCount(5);
        statusArea.setFont(Font.font("Monospaced", 11));
        statusArea.setText("Sistema listo. Configure los parámetros y presione 'Guardar Configuración'.\n");

        // Ensamblar layout
        mainLayout.setCenter(tabPane);
        mainLayout.setBottom(new VBox(10, buttonPanel, statusArea));

        primaryStage.setScene(scene);

    /**

        ipTextField.setPrefWidth(200);
        ipBox.getChildren().addAll(ipLabel, ipTextField);

        portBox.setAlignment(Pos.CENTER_LEFT);
        Label portLabel = new Label("Puerto:");
        portLabel.setPrefWidth(150);
        portTextField = new TextField("5084");
        portTextField.setPrefWidth(200);
        portBox.getChildren().addAll(portLabel, portTextField);

        VBox formatBox = new VBox(10);
        Label formatLabel = new Label("Formato de EPC:");
        formatLabel.setFont(Font.font("System", FontWeight.BOLD, 12));

        epcFormatGroup = new ToggleGroup();
        hexRadio = new RadioButton("Hexadecimal (recomendado)");
        hexRadio.setToggleGroup(epcFormatGroup);

    /**
     * Crea el panel de configuración de antenas
     */
    private ScrollPane createAntennasPanel() {
        VBox panel = new VBox(15);
        panel.setPadding(new Insets(20));

        // Título
        Label titleLabel = new Label("Configuración de Antenas (1-4)");
        titleLabel.setFont(Font.font("System", FontWeight.BOLD, 16));

        // RSSI de corte global
        VBox rssiGlobalBox = new VBox(10);
        Label rssiGlobalLabel = new Label("RSSI de Corte Global:");
        rssiGlobalLabel.setFont(Font.font("System", FontWeight.BOLD, 12));

        HBox rssiSliderBox = new HBox(10);
        rssiSliderBox.setAlignment(Pos.CENTER_LEFT);
        rssiCutoffSlider = new Slider(-90, -30, -70);
        rssiCutoffSlider.setShowTickLabels(true);
        rssiCutoffSlider.setShowTickMarks(true);
        rssiCutoffSlider.setMajorTickUnit(10);
        rssiCutoffSlider.setMinorTickCount(5);
        rssiCutoffSlider.setPrefWidth(400);

        rssiCutoffLabel = new Label("-70 dBm");
        rssiCutoffLabel.setPrefWidth(80);
        rssiCutoffLabel.setStyle("-fx-font-weight: bold;");

        rssiCutoffSlider.valueProperty().addListener((obs, oldVal, newVal) -> {
            rssiCutoffLabel.setText(String.format("%.0f dBm", newVal.doubleValue()));
        });

        rssiSliderBox.getChildren().addAll(rssiCutoffSlider, rssiCutoffLabel);
        rssiGlobalBox.getChildren().addAll(rssiGlobalLabel, rssiSliderBox);

        panel.getChildren().addAll(titleLabel, new Separator(), rssiGlobalBox, new Separator());

        // Inicializar arrays
        antennaCheckboxes = new CheckBox[4];
        txPowerSliders = new Slider[4];
        txPowerLabels = new Label[4];
        rxSensitivitySliders = new Slider[4];
        rxSensitivityLabels = new Label[4];

        // Crear configuración para cada antena
        for (int i = 0; i < 4; i++) {
            panel.getChildren().add(createAntennaConfig(i + 1));
        }

        // Botón para aplicar configuración a todas
        Button applyAllButton = new Button("Aplicar Config a Todas las Antenas");
        applyAllButton.setOnAction(e -> applyToAllAntennas());
        applyAllButton.setStyle("-fx-background-color: #FF9800; -fx-text-fill: white;");

        panel.getChildren().add(applyAllButton);

        ScrollPane scrollPane = new ScrollPane(panel);
        scrollPane.setFitToWidth(true);
        return scrollPane;
    }

    /**
     * Crea la configuración para una antena individual
     */
    private VBox createAntennaConfig(int antennaNumber) {
        int index = antennaNumber - 1;
        VBox antennaBox = new VBox(10);
        antennaBox.setPadding(new Insets(10));
        antennaBox.setStyle("-fx-border-color: #CCCCCC; -fx-border-width: 1; -fx-border-radius: 5;");

        // Checkbox de habilitación
        antennaCheckboxes[index] = new CheckBox("Antena " + antennaNumber + " - Habilitada");
        antennaCheckboxes[index].setSelected(true);
        antennaCheckboxes[index].setFont(Font.font("System", FontWeight.BOLD, 12));

        // TX Power
        Label txLabel = new Label("Potencia TX (dBm):");
        HBox txBox = new HBox(10);
        txBox.setAlignment(Pos.CENTER_LEFT);

        txPowerSliders[index] = new Slider(10, 32.5, 30);
        txPowerSliders[index].setShowTickMarks(true);
        txPowerSliders[index].setMajorTickUnit(5);
        txPowerSliders[index].setPrefWidth(300);

        txPowerLabels[index] = new Label("30.0 dBm");
        txPowerLabels[index].setPrefWidth(80);

        int finalIndex = index;
        txPowerSliders[index].valueProperty().addListener((obs, oldVal, newVal) -> {
            txPowerLabels[finalIndex].setText(String.format("%.1f dBm", newVal.doubleValue()));
        });

        txBox.getChildren().addAll(txPowerSliders[index], txPowerLabels[index]);

        // RX Sensitivity
        Label rxLabel = new Label("Sensibilidad RX (dBm):");
        HBox rxBox = new HBox(10);
        rxBox.setAlignment(Pos.CENTER_LEFT);

        rxSensitivitySliders[index] = new Slider(-90, -50, -70);
        rxSensitivitySliders[index].setShowTickMarks(true);
        rxSensitivitySliders[index].setMajorTickUnit(10);
        rxSensitivitySliders[index].setPrefWidth(300);

        rxSensitivityLabels[index] = new Label("-70 dBm");
        rxSensitivityLabels[index].setPrefWidth(80);

        rxSensitivitySliders[index].valueProperty().addListener((obs, oldVal, newVal) -> {
            rxSensitivityLabels[finalIndex].setText(String.format("%.0f dBm", newVal.doubleValue()));
        });

        rxBox.getChildren().addAll(rxSensitivitySliders[index], rxSensitivityLabels[index]);

        antennaBox.getChildren().addAll(
            antennaCheckboxes[index],
            txLabel,
            txBox,
            rxLabel,
            rxBox
        );

        return antennaBox;
    }

    /**
     * Aplica la configuración de la primera antena a todas
     */
    private void applyToAllAntennas() {

        for (int i = 1; i < 4; i++) {
            txPowerSliders[i].setValue(txPower);
            rxSensitivitySliders[i].setValue(rxSensitivity);
        }

        appendStatus("✓ Configuración aplicada a todas las antenas\n");
    }

    /**
     * Guarda la configuración actual
     */
    private void saveConfiguration() {
        try {
            // Configuración de red
            config.setProperty("reader.ip", ipTextField.getText());
            config.setProperty("reader.port", portTextField.getText());
            config.setProperty("epc.format", hexRadio.isSelected() ? "HEX" : "DEC");

            // Configuración de API
            config.setProperty("api.endpoint", apiEndpointTextField.getText());
            config.setProperty("api.key", apiKeyField.getText());

            // RSSI de corte
            config.setProperty("rssi.cutoff", String.valueOf(rssiCutoffSlider.getValue()));

            // Configuración de antenas
            for (int i = 0; i < 4; i++) {
                config.setProperty("antenna." + (i+1) + ".enabled", 
                    String.valueOf(antennaCheckboxes[i].isSelected()));
                config.setProperty("antenna." + (i+1) + ".txPower", 
                    String.valueOf(txPowerSliders[i].getValue()));
                config.setProperty("antenna." + (i+1) + ".rxSensitivity", 
                    String.valueOf(rxSensitivitySliders[i].getValue()));
            }

            // Configuración avanzada
            config.setProperty("search.mode", searchModeCombo.getValue());
            config.setProperty("session", String.valueOf(sessionCombo.getValue()));
            config.setProperty("reader.mode", readerModeCombo.getValue());

            // Guardar a archivo
            try (FileOutputStream out = new FileOutputStream(CONFIG_FILE)) {
                config.store(out, "Configuración RFID - Zebra FX7500");
            }

            appendStatus("✓ Configuración guardada exitosamente en " + CONFIG_FILE + "\n");

        } catch (IOException e) {
            appendStatus("✗ Error guardando configuración: " + e.getMessage() + "\n");
        }
    }

    /**
     * Carga la configuración desde archivo
     */
    private void loadConfiguration() {
        try (FileInputStream in = new FileInputStream(CONFIG_FILE)) {
            config.load(in);
            applyLoadedConfiguration();
            appendStatus("✓ Configuración cargada desde " + CONFIG_FILE + "\n");
        } catch (IOException e) {
            appendStatus("✗ No se pudo cargar la configuración: " + e.getMessage() + "\n");
            loadDefaultConfig();
        }
    }

    /**
     * Aplica la configuración cargada a los componentes
     */
    private void applyLoadedConfiguration() {
        // Configuración de red
        ipTextField.setText(config.getProperty("reader.ip", "192.168.1.100"));
        portTextField.setText(config.getProperty("reader.port", "5084"));

        if ("HEX".equals(config.getProperty("epc.format", "HEX"))) {
            hexRadio.setSelected(true);
        } else {
            decimalRadio.setSelected(true);
        }

        // Configuración de API
        apiEndpointTextField.setText(config.getProperty("api.endpoint", 
            "http://localhost:8080/api/rfid/tags"));
        apiKeyField.setText(config.getProperty("api.key", ""));

        // RSSI de corte
        rssiCutoffSlider.setValue(Double.parseDouble(
            config.getProperty("rssi.cutoff", "-70")));

        // Configuración de antenas
        for (int i = 0; i < 4; i++) {
            antennaCheckboxes[i].setSelected(Boolean.parseBoolean(
                config.getProperty("antenna." + (i+1) + ".enabled", "true")));
            txPowerSliders[i].setValue(Double.parseDouble(
                config.getProperty("antenna." + (i+1) + ".txPower", "30")));
            rxSensitivitySliders[i].setValue(Double.parseDouble(
                config.getProperty("antenna." + (i+1) + ".rxSensitivity", "-70")));
        }

        // Configuración avanzada
        searchModeCombo.setValue(config.getProperty("search.mode", "DualTarget"));
        sessionCombo.setValue(Integer.parseInt(config.getProperty("session", "2")));
        readerModeCombo.setValue(config.getProperty("reader.mode", "AutoSetDenseReader"));
    }

    /**
     * Carga la configuración por defecto
     */
    private void loadDefaultConfig() {
        ipTextField.setText("192.168.1.100");
        portTextField.setText("5084");
        hexRadio.setSelected(true);
        apiEndpointTextField.setText("http://localhost:8080/api/rfid/tags");
        apiKeyField.setText("");
        rssiCutoffSlider.setValue(-70);

        for (int i = 0; i < 4; i++) {
            antennaCheckboxes[i].setSelected(true);
            txPowerSliders[i].setValue(30);
            rxSensitivitySliders[i].setValue(-70);
        }

        searchModeCombo.setValue("DualTarget");
        sessionCombo.setValue(2);
        readerModeCombo.setValue("AutoSetDenseReader");

        appendStatus("✓ Configuración restaurada a valores por defecto\n");
    }

    /**
     * Inicia la lectura RFID con la configuración actual
     */
    private void startReading() {
        // Guardar configuración antes de iniciar
        saveConfiguration();

        appendStatus("\n=== INICIANDO LECTURA RFID ===\n");
        appendStatus("Configuración:\n");
        appendStatus("- Lector: " + ipTextField.getText() + ":" + portTextField.getText() + "\n");
        appendStatus("- API: " + apiEndpointTextField.getText() + "\n");
        appendStatus("- Formato EPC: " + (hexRadio.isSelected() ? "Hexadecimal" : "Decimal") + "\n");
        appendStatus("- RSSI Corte: " + String.format("%.0f dBm\n", rssiCutoffSlider.getValue()));

        int enabledAntennas = 0;
        for (int i = 0; i < 4; i++) {
            if (antennaCheckboxes[i].isSelected()) {
                enabledAntennas++;
                appendStatus(String.format("- Antena %d: TX=%.1f dBm, RX=%.0f dBm\n", 
                    i+1, txPowerSliders[i].getValue(), rxSensitivitySliders[i].getValue()));
            }
        }
        appendStatus("- Total antenas habilitadas: " + enabledAntennas + "\n");

        appendStatus("\n▶ Lanzando aplicación principal...\n");

        // Aquí se lanzaría la aplicación principal ZebraRFIDConnectorApp
        // con los parámetros configurados
        new Thread(() -> {
            try {
                ZebraRFIDConnectorApp.launchWithConfig(config);
            } catch (Exception e) {
                Platform.runLater(() -> 
                    appendStatus("✗ Error al iniciar: " + e.getMessage() + "\n"));
            }
        }).start();
    }

    /**
     * Agrega texto al área de status
     */
    private void appendStatus(String message) {
        Platform.runLater(() -> {
            statusArea.appendText(message);
            statusArea.setScrollTop(Double.MAX_VALUE);
        });
    }

    public static void main(String[] args) {
        launch(args);
    }

    /**
     * Obtiene la configuración actual
     */
    public Properties getConfig() {
        return config;
    }
